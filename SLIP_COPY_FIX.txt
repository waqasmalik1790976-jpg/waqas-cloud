/* SLIP COPY FIX
   Replace ONLY the existing slipCopy() function in index.html
   with the function below.
*/

async function slipCopy(i,btn){

  if(!window.html2canvas){
    alert('Library did not load.');
    return;
  }

  document.body.classList.add('cap');

  try{

    var card = L.children[i].querySelector('.card');

    var cv = await html2canvas(card,{
      scale:1.75,
      backgroundColor:null,
      useCORS:true,
      allowTaint:true,
      logging:false,

      onclone:function(d){

        /* Do not crop the scroll container */
        d.querySelectorAll('.scroll').forEach(function(x){
          x.style.overflow='visible';
          x.style.height='auto';
          x.style.maxHeight='none';
        });

        /* IMPORTANT: remove the old top margin/padding that caused
           the bank heading to be clipped during copy */
        d.querySelectorAll('.card').forEach(function(x){
          x.style.overflow='visible';
          x.style.marginTop='0px';
          x.style.paddingTop='0px';
          x.style.height='auto';
          x.style.maxHeight='none';
          x.style.boxSizing='border-box';
        });

        /* Keep bank/party/petrol headers fully inside the canvas */
        d.querySelectorAll('.hd,.ph,.gh').forEach(function(x){
          x.style.overflow='visible';
          x.style.height='auto';
          x.style.minHeight='0';
          x.style.boxSizing='border-box';
        });

        /* Keep heading text visible */
        d.querySelectorAll('.bank,.pt,.gt').forEach(function(x){
          x.style.overflow='visible';
          x.style.height='auto';
          x.style.minHeight='0';
          x.style.whiteSpace='normal';
          x.style.wordBreak='normal';
          x.style.lineHeight='1.25';
        });

        /* Small safe top space so the heading can never touch
           the capture boundary */
        var c=d.querySelector('.card');
        if(c){
          c.style.paddingTop='8px';
        }
      }
    });

    cv=withBg(cv);

    var blob=await new Promise(function(resolve){
      cv.toBlob(resolve,'image/png');
    });

    try{
      await navigator.clipboard.write([
        new ClipboardItem({
          'image/png':blob
        })
      ]);

      flash(btn,'✓ Copied');

    }catch(e){

      var a=document.createElement('a');
      a.href=URL.createObjectURL(blob);
      a.download='slip-'+(i+1)+'.png';
      a.click();

      flash(btn,'✓ Saved');
    }

  }catch(err){

    console.error(err);
    alert('Could not copy the slip.');

  }

  document.body.classList.remove('cap');
}
